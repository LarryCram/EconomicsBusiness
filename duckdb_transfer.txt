  SELECT "Journal name" AS journal_name, e.issn, j.id AS source_id
  FROM econ_bus.incites_econ_bus e
  LEFT JOIN 'D:\openalex_nov25\parquet\sources.parquet' j
  ON List_contains(j.issn, e.issn) -- OR list_contains(j.issn, e.eissn)

  )

CREATE OR REPLACE TABLE econ_bus.works_20_25 AS
SELECT *
  FROM 
    (SELECT idx AS work_idx, source_id[23:]::BIGINT AS source_idx,
            work_id, title, source_id, source.display_name AS source_name, doi, publication_year,
            authors_count, institutions_distinct_count, 
            referenced_works_count, cited_by_count,
            "type", is_retracted, is_paratext
      FROM econ_bus.incites_oa i
      LEFT JOIN 'D:\openalex_nov25\parquet\works\*.parquet' j
      ON j.source.id = i.source_id AND (publication_year BETWEEN 2020 AND 2025)
      WHERE is_retracted=false AND is_paratext=false AND list_contains(['article', 'review'], "type")
      -- LIMIT 256
      );

-- Create work institution - weight tuples
-- =======================================
CREATE OR REPLACE TABLE econ_bus.institution_weights AS
  SELECT idx AS work_idx, 
          institution_id[23:]::BIGINT AS institution_idx,
          1.0/count(DISTINCT idx) AS weight,
    FROM 'D:\openalex_nov25\parquet\authorships\*.parquet' a
    GROUP BY idx, institution_id
    LIMIT 128

  KILLER

  -- JOIN institution weights to works and sources
